import { Injectable, NotFoundException, ForbiddenException } from '@nestjs/common';
import { InventoryRepository } from './inventory.repository';
import { CreateProductDto, UpdateProductDto } from '@ryzera/pos-schema';
import { BranchScopeResult } from '../auth/decorators/branch-scope.decorator'; // adjust path

@Injectable()
export class InventoryService {
    constructor(private readonly repo: InventoryRepository) {}

    async findAll(companyId: number | undefined, branchId: number | undefined, scope: BranchScopeResult) {
        // scope.branchId is undefined for Admin (no restriction).
        // For everyone else, it OVERRIDES whatever branchId was passed in the query â€”
        // a Manager can't page ?branchId=5 to peek at another branch.
        const effectiveBranchId = scope.branchId ?? branchId;
        return this.repo.findAll(companyId, effectiveBranchId);
    }

    async findOne(id: number, scope: BranchScopeResult) {
        const product = await this.repo.findById(id);
        if (!product) throw new NotFoundException(`Product #${id} not found`);

        // Manager/Staff trying to view a product outside their branch â€” pretend it
        // doesn't exist rather than leaking a 403 (don't confirm other branches' data).
        if (scope.branchId && product.branch_id !== scope.branchId) {
            throw new NotFoundException(`Product #${id} not found`);
        }
        return product;
    }

    async create(dto: CreateProductDto, scope: BranchScopeResult) {
        // Non-admins can only create products in their own branch â€” force it,
        // ignore/override anything the client sent for branch_id.
        const branch_id = scope.branchId ?? (dto as any).branch_id;
        return this.repo.create({ ...dto, branch_id } as CreateProductDto);
    }

    async update(id: number, dto: UpdateProductDto, scope: BranchScopeResult) {
        const existing = await this.findOne(id, scope); // reuses the branch check above

        // Also block a Manager from moving a product to a different branch via update.
        if (scope.branchId && (dto as any).branch_id && (dto as any).branch_id !== scope.branchId) {
            throw new ForbiddenException('You cannot move a product to a different branch.');
        }

        return this.repo.update(id, dto);
    }

    async remove(id: number) {
        const product = await this.repo.findById(id);
        if (!product) throw new NotFoundException(`Product #${id} not found`);
        return this.repo.delete(id);
    }
}
